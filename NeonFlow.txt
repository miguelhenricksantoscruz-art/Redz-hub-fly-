local connections, threads = {}, {}

-- safe invoke wrapper (tries multiple names)
local function safeInvokeRemote(name, ...)
    local ok, res = pcall(function() 
        local rem = ReplicatedStorage:FindFirstChild(name)
        if rem and rem:IsA("RemoteFunction") then return rem:InvokeServer(...) end
    end)
    if ok then return res end
    return nil
end

local function safeFireRemote(name, ...)
    pcall(function()
        local rem = ReplicatedStorage:FindFirstChild(name)
        if rem and rem:IsA("RemoteEvent") then rem:FireServer(...) end
    end)
end

--==================== Tween / Movimento ======================--
local TweenSPS, ActiveTween = 140, nil
local function stopTween() if ActiveTween then pcall(function() ActiveTween:Cancel() end) ActiveTween=nil end end
local function tweenTo(cf)
    if not cf then return end
    local root = HRP()
    local dist = (root.Position - cf.Position).Magnitude
    local t = math.clamp(dist / math.max(10, TweenSPS), 0.25, 3.0)
    local info = TweenInfo.new(t, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut)
    stopTween()
    ActiveTween = TweenService:Create(root, info, {CFrame = cf})
    ActiveTween:Play(); ActiveTween.Completed:Wait()
    ActiveTween = nil
end

--==================== Spawn / Travel =========================--
local function setSpawn()
    pcall(function()
        local comm = ReplicatedStorage:FindFirstChild("Remotes") and ReplicatedStorage.Remotes:FindFirstChild("CommF_")
        if comm and comm:IsA("RemoteFunction") then comm:InvokeServer("SetSpawnPoint") end
    end)
end

local function travelToSea(sea)
    pcall(function()
        local rem = ReplicatedStorage:FindFirstChild("Remotes") and ReplicatedStorage.Remotes:FindFirstChild("CommF_")
        if not rem then return end
        if sea == 1 then rem:InvokeServer("TravelMain")
        elseif sea == 2 then rem:InvokeServer("TravelDressrosa")
        elseif sea == 3 then rem:InvokeServer("TravelZou") end
    end)
end

--==================== Player data helpers =====================--
local function getLevel()
    local ok, v = pcall(function() return LP:WaitForChild("Data"):WaitForChild("Level").Value end)
    if ok and v then return v end
    local ok2, v2 = pcall(function() return LP:FindFirstChild("PlayerGui") and 1 end)
    if ok2 then return v2 or 1 end
    return 1
end

local function getStatPoints()
    -- tenta múltiplos possíveis nomes
    local candidates = {"StatPoints","Points","SP","Stat_Point"}
    for _,name in ipairs(candidates) do
        local s = LP:FindFirstChild("Data") and LP.Data:FindFirstChild(name)
        if s and s.Value then
            return s.Value
        end
    end
    -- fallback
    local ok, v = pcall(function() return LP:FindFirstChild("Data") and LP.Data:FindFirstChild("StatPoints") and LP.Data.StatPoints.Value end)
    if ok and v then return v end
    return 0
end

    local list = weaponList[SelectedCategory]
    if list then
        for _,name in ipairs(list) do
            local tool = Backpack:FindFirstChild(name) or Ch:FindFirstChild(name)

        end
    end
    for _,tool in ipairs(Backpack:GetChildren()) do
        if tool:IsA("Tool") then tool.Parent = Ch return true end
    end

--================== Busca de inimigos ========================--
local EnemiesFolder = Workspace:FindFirstChild("Enemies") or Workspace:WaitForChild("Enemies", 5)
local function validEnemyModel(m)
    if not m or not m:IsA("Model") then return false end
    local h = m:FindFirstChild("Humanoid")
    local r = m:FindFirstChild("HumanoidRootPart")
    return h and r and h.Health and h.Health > 0
end

local function findEnemyByNamePart(namePart)
    local lower = string.lower(namePart or "")
    local container = EnemiesFolder or Workspace
    for _,mob in ipairs(container:GetChildren()) do
        if validEnemyModel(mob) and string.find(string.lower(mob.Name), lower) then
            return mob
        end
    end
    -- fallback: search deep
    for _,mob in ipairs(Workspace:GetDescendants()) do
        if mob:IsA("Model") and validEnemyModel(mob) and string.find(string.lower(mob.Name), lower) then
            return mob
        end
    end
    return nil
end

--================== Hitbox + FastAttack =====================--
local HitboxSize = 70
local function applyBigHitbox(target)
    if not target then return end
    local hrp = target:FindFirstChild("HumanoidRootPart"); if not hrp then return end
    pcall(function()
        hrp.CanCollide = false
        hrp.Size = Vector3.new(HitboxSize, HitboxSize, HitboxSize)
        hrp.Massless = true
    end)
end

local FastAttackActive = false
local function fastAttackTick()
    if not FastAttackActive then return end
    local hum = Hum()
    pcall(function()
        if hum and hum:FindFirstChild("Animator") then
            for _,track in ipairs(hum.Animator:GetPlayingAnimationTracks()) do
                if track and track.IsPlaying then track:AdjustSpeed(1.2) end
            end
        end
    end)
end

--================== AUTO CLICK (estilo app) ===================
-- Modes: "Fixed", "Center", "OnTarget"
local AC = {
    Enabled = false,
    CPS = 12,
    Mode = "Fixed",
    X = 640, Y = 360,
    Hotkey = Enum.KeyCode.F6,
    CurrentTarget = nil,
}
local function screenClick(x, y)
    if not x or not y then return end
    pcall(function()
        Vim:SendMouseButtonEvent(x, y, 0, true, game, 0)
        Vim:SendMouseButtonEvent(x, y, 0, false, game, 0)
    end)
end
local function getCenter()
    local v = Cam.ViewportSize
    return math.floor(v.X/2), math.floor(v.Y/2)
end
local function getTargetScreenXY()
    local tgt = AC.CurrentTarget
    if not (tgt and tgt.Parent) then return nil end
    local hrp = tgt:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local v3, onscreen = Cam:WorldToViewportPoint(hrp.Position)
    if not onscreen then return nil end
    return math.floor(v3.X), math.floor(v3.Y)
end

stopThread("autoclick")
threads.autoclick = task.spawn(function()
    while true do
        if AC.Enabled then
            local x, y
            if AC.Mode == "Center" then
                x, y = getCenter()
            elseif AC.Mode == "OnTarget" then
                x, y = getTargetScreenXY()
                if not x then x, y = getCenter() end
            else
                x, y = AC.X, AC.Y
            end
            screenClick(x, y)
            -- fallback: Tool activate + VirtualUser
            pcall(function()
                local tool = Char():FindFirstChildOfClass("Tool")
                if tool then tool:Activate() end
            end)
            pcall(function()
                VirtualUser:CaptureController()
                VirtualUser:Button1Down(Vector2.new(0,0))

local HoverEnabled, HoverHeight = true, 18

--====================== Farm Core ============================--
local AutoFarmEnabled, stopFarmSignal = false, false

local function takeQuest(q)
    if not q then return end
    travelToSea(q.sea)
    if q.qpos then tweenTo(q.qpos) end
    setSpawn()
    task.wait(0.15)
    pcall(function()
        local rem = ReplicatedStorage:FindFirstChild("Remotes") and ReplicatedStorage.Remotes:FindFirstChild("CommF_")
        if rem and rem:IsA("RemoteFunction") then rem:InvokeServer("StartQuest", q.quest, 1) end
    end)
end

local function hunt(q)
    equipWeapon()
    local target = findEnemyByNamePart(q.npc)
    if not target then if q.epos and #q.epos>0 then tweenTo(q.epos[1]) end return end
    local r2 = target:FindFirstChild("HumanoidRootPart")
    local h2 = target:FindFirstChild("Humanoid")
    if not (r2 and h2) then return end

    applyBigHitbox(target)
    AC.CurrentTarget = target

    local above = r2.CFrame * CFrame.new(0, HoverHeight, 0)
    tweenTo(above)

    repeat
        if stopFarmSignal then break end
        if HoverEnabled then
            pcall(function() HRP().CFrame = r2.CFrame * CFrame.new(0, HoverHeight, 0) end)
        else
            pcall(function() HRP().CFrame = r2.CFrame * CFrame.new(0,0,3) end)
        end

        applyBigHitbox(target)
        FastAttackActive = true
        fastAttackTick()

        RunService.Heartbeat:Wait()
    until (not h2) or (h2.Health <= 0) or not target.Parent or not AutoFarmEnabled

    FastAttackActive = false
    AC.CurrentTarget = nil
end

local function questForLevel(lvl)
    -- retorna a primeira matching quest (corrigido para evitar pegar a última)
    for _,q in ipairs(QuestDB) do
        if lvl >= q.min and (not q.max or lvl <= q.max) then
            return q
        end
    end
    return nil
end

local function farmLoop()
    stopFarmSignal = false
    while AutoFarmEnabled and not stopFarmSignal do
        local lvl = getLevel()
        local q = questForLevel(lvl)
        if not q then task.wait(0.25) goto cont end
        travelToSea(q.sea)
        if not (LP.PlayerGui and LP.PlayerGui.Main and LP.PlayerGui.Main:FindFirstChild("Quest") and LP.PlayerGui.Main.Quest.Visible) then
            takeQuest(q)
            task.wait(0.8)
        end
        hunt(q)
        ::cont::
        RunService.Heartbeat:Wait()
    end
end

local AutoStats = {
    Enabled = false,
    Mode = "Melee", -- "Melee","Sword","Gun","Blox Fruta"
    Delay = 0.35,
}
-- tenta vários remotes/nomes para adicionar stat
local function tryAddStat(stat)
    -- nomes de remotes comuns: "AddPoint","AddStat","BuyStat","BuyStats"
    local tries = {
        function() local rem = ReplicatedStorage:FindFirstChild("Remotes") and ReplicatedStorage.Remotes:FindFirstChild("AddPoint"); if rem and rem:IsA("RemoteFunction") then rem:InvokeServer(stat) end end,
        function() local rem = ReplicatedStorage:FindFirstChild("Remotes") and ReplicatedStorage.Remotes:FindFirstChild("AddStat"); if rem and rem:IsA("RemoteFunction") then rem:InvokeServer(stat) end end,
        function() local rem = ReplicatedStorage:FindFirstChild("Remotes") and ReplicatedStorage.Remotes:FindFirstChild("BuyStat"); if rem and rem:IsA("RemoteFunction") then rem:InvokeServer(stat) end end,
        function() local rem = ReplicatedStorage:FindFirstChild("Remotes") and ReplicatedStorage.Remotes:FindFirstChild("BuyStats"); if rem and rem:IsA("RemoteFunction") then rem:InvokeServer(stat) end end,
    }
    for _,fn in ipairs(tries) do pcall(fn) end
end

stopThread("autostats")
threads.autostats = nil
local function startAutoStats()
    stopThread("autostats")
    threads.autostats = task.spawn(function()
        while AutoStats.Enabled do
            local pts = getStatPoints()
            if pts <= 0 then
                task.wait(1)
            else
                local statName
                if AutoStats.Mode == "Melee" then statName = "Melee"
                elseif AutoStats.Mode == "Sword" then statName = "Sword"
                elseif AutoStats.Mode == "Gun" then statName = "Gun"
                else statName = "Fruit" end
                -- tentar aplicar enquanto tiver pontos
                while getStatPoints() > 0 and AutoStats.Enabled do
                    tryAddStat(statName)
                    task.wait(AutoStats.Delay)
                end
            end
            task.wait(0.5)
        end
    end)
end
