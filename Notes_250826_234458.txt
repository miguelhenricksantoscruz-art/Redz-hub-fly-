local plrs = game:GetService("Players")
local lp = plrs.LocalPlayer
local rs = game:GetService("RunService")
local ts = game:GetService("TweenService")
local rep = game:GetService("ReplicatedStorage")
local vu = game:GetService("VirtualUser")

local function SafeHRP()
    local ch = lp.Character
    if ch and ch:FindFirstChild("HumanoidRootPart") then
        return ch.HumanoidRootPart
    end
end

local function WTS(cf, speed)
    local hrp = SafeHRP()
    if not hrp then return end
    speed = speed or 250
    local dist = (hrp.Position - cf.Position).Magnitude
    if dist < speed then
        hrp.CFrame = cf
    else
        local t = ts:Create(hrp, TweenInfo.new(dist/speed, Enum.EasingStyle.Linear), {CFrame = cf})
        t:Play()
        t.Completed:Wait()
    end
end

local function EquipTool(nameList)
    for _,v in next, lp.Backpack:GetChildren() do
        if v:IsA("Tool") and (not nameList or nameList[v.Name]) then
            lp.Character.Humanoid:EquipTool(v)
            return v
        end
    end
    for _,v in next, lp.Character:GetChildren() do
        if v:IsA("Tool") and (not nameList or nameList[v.Name]) then
            return v
        end
    end
end

local function ClickAttack()
    local tool = EquipTool()
    if tool and tool:FindFirstChild("RemoteFunction") then
        pcall(function()
            tool.RemoteFunction:InvokeServer("Z")
        end)
    else
        vu:CaptureController()
        vu:ClickButton1(Vector2.new(0,0))
    end
end

local function BringMob(model, pos)
    pcall(function()
        for _,m in next, model:GetChildren() do
            if m:IsA("Model") and m:FindFirstChild("HumanoidRootPart") and m:FindFirstChild("Humanoid") then
                m:FindFirstChild("Humanoid").Sit = false
                m:FindFirstChild("HumanoidRootPart").Size = Vector3.new(60,60,60)
                m:FindFirstChild("HumanoidRootPart").CFrame = pos
                m:FindFirstChild("Humanoid").WalkSpeed = 0
                m:FindFirstChild("Humanoid").JumpPower = 0
            end
        end
    end)
end

local function GetMobFolder()
    local ws = workspace
    for _,v in next, ws:GetChildren() do
        if v:IsA("Folder") and (v.Name == "Enemies" or v.Name == "Enemys" or v.Name == "MOB" or v.Name == "Monsters") then
            return v
        end
    end
    return ws:FindFirstChild("Enemies") or ws
end

local function GetQuestUI()
    for _,v in next, workspace:GetDescendants() do
        if v:IsA("ProximityPrompt") and v.Parent and v.Parent.Name:lower():find("quest") then
            return v.Parent
        end
    end
end

local function TakeQuest()
    CheckQuest()
    if not CFrameQuest then return end
    WTS(CFrameQuest, 300)
    task.wait(0.2)
    pcall(function()
        rep.Remotes.CommF_:InvokeServer("StartQuest", NameQuest, LevelQuest)
    end)
end

local function IsQuestTaken()
    pcall(function()
        local q = lp.PlayerGui and lp.PlayerGui.Main and lp.PlayerGui.Main.Quest
        if q and q.Visible then
            QuestTaken = true
        else
            QuestTaken = false
        end
    end)
    return QuestTaken
end

local function FarmStep()
    if not _G.NM_AutoFarm then return end
    CheckQuest()
    if not IsQuestTaken() then
        TakeQuest()
        return
    end
    local mobFolder = GetMobFolder()
    local target = nil
    for _, m in next, mobFolder:GetChildren() do
        if m:IsA("Model") and m:FindFirstChild("Humanoid") and m:FindFirstChild("HumanoidRootPart") then
            if m.Name:find(NameMon) and m.Humanoid.Health > 0 then
                target = m
                break
            end
        end
    end
    if target then
        local pos = target.HumanoidRootPart.CFrame * CFrame.new(0, 18, 0)
        BringMob(target.Parent or mobFolder, pos)
        WTS(pos, 250)
        AutoBuso()
        ClickAttack()
    else
        -- sem alvo visível, volta pro spawn dos mobs
        if CFrameMon then
            WTS(CFrameMon, 320)
        end
    end
end
